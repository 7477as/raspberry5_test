cmake_minimum_required(VERSION 3.16)
project(spps VERSION 1.0.0 LANGUAGES C CXX)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-unix-2.0)
add_executable(spps Main.cpp SppServer.cpp PacketCodec.cpp LatencyBenchmark.cpp)
target_link_libraries(spps PRIVATE PkgConfig::GIO)
target_compile_options(spps PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    add_executable(spps_test BenchmarkTest.cpp PacketCodec.cpp LatencyBenchmark.cpp)
    target_link_libraries(spps_test PRIVATE Threads::Threads)
    target_compile_options(spps_test PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME protocol_and_benchmark COMMAND spps_test)
    set_tests_properties(protocol_and_benchmark PROPERTIES TIMEOUT 20)
endif()
