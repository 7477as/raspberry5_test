cmake_minimum_required(VERSION 3.16)
project(gatts VERSION 1.0.0 LANGUAGES C CXX)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-unix-2.0)
find_package(Threads REQUIRED)
add_executable(gatts Main.cpp GattServer.cpp LatencyBenchmark.cpp)
target_link_libraries(gatts PRIVATE PkgConfig::GIO Threads::Threads)
target_compile_options(gatts PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(gatts_test BenchmarkTest.cpp LatencyBenchmark.cpp)
    target_link_libraries(gatts_test PRIVATE PkgConfig::GIO Threads::Threads)
    target_compile_options(gatts_test PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME gatt_protocol_benchmark COMMAND gatts_test)
    set_tests_properties(gatt_protocol_benchmark PROPERTIES TIMEOUT 20)
endif()
